\begin{frame}{Preguntas que la visualización debe responder}
\small
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{itemize}
  \item ¿Qué tan reciente y disponible es la evidencia visual en cada zona?
  \item ¿Qué elementos peatonales y viales aparecen con mayor frecuencia por 100 imágenes?
  \item ¿Dónde se concentran cruces, veredas, señales, iluminación y marcas viales?
\end{itemize}
\column{0.49\textwidth}
\begin{itemize}
  \item ¿Qué contrastes persisten entre zonas equivalentes?
  \item ¿Qué imagen o secuencia explica cada patrón agregado?
  \item ¿Qué sesgos o vacíos impiden una conclusión fuerte?
\end{itemize}
\end{columns}
\vspace{0.45cm}
\colorbox{navy}{\parbox[c][1.1cm][c]{0.98\linewidth}{\centering\color{white}\large La interfaz conectará el mapa, los filtros comparativos y la evidencia fotográfica.}}
\end{frame}
